#OBJECT TABLE
CREATE OR REPLACE TYPE student_obj AS 
OBJECT (
    student_id NUMBER,
    name VARCHAR2(50),
    age NUMBER
);
/
CREATE TABLE student_table OF student_obj;

#OBJECT COLUMN
CREATE OR REPLACE TYPE address_obj AS OBJECT (
    city VARCHAR2(50),
    country VARCHAR2(50)
);
/
CREATE TABLE customer (
    cust_id NUMBER,
    cust_name VARCHAR2(50),
    address address_obj
);

/

    courses course_nt
